CREATE TABLE ghe_funcao (
    id SERIAL PRIMARY KEY,
    unidade VARCHAR(200),
    ghe VARCHAR(10),
    setor VARCHAR(100),
    funcao VARCHAR(150)
);



CREATE TABLE ghe_epi (
    unidade VARCHAR(200),
    ghe VARCHAR(10),
    epi VARCHAR(200)
);



CREATE TABLE epi_entregue (
    id SERIAL PRIMARY KEY,
    
    unidade VARCHAR(200) NOT NULL,
    
    nome VARCHAR(150) NOT NULL,
    funcao VARCHAR(100) NOT NULL,
    setor VARCHAR(100) NOT NULL,
    
    epi VARCHAR(150) NOT NULL,
    
    data_entrega DATE NOT NULL
);
